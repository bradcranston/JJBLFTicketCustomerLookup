FileMaker Script Examples:

To load data from FileMaker, use this JavaScript in your WebViewer:

// Option 1: Direct array format
var customerData = [
    {
        "CustomerNumber": "123",
        "Customer": "ABC Company",
        "Name": "Main Office",
        "AddrServ_1Line": "123 Main St, City, State 12345"
    }
    // ... more records
];
window.loadCustomerData(customerData);

// Option 2: OData format (like your current data.json)
var odataResponse = {
    "value": [
        {
            "CustomerNumber": "123",
            "Customer": "ABC Company", 
            "Name": "Main Office",
            "AddrServ_1Line": "123 Main St, City, State 12345"
        }
        // ... more records
    ]
};
window.loadCustomerData(odataResponse);

// Option 3: JSON string format
var jsonString = '{"value":[{"CustomerNumber":"123",...}]}';
window.loadCustomerData(jsonString);
